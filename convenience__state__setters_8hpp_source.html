109inline void setContinuousStateCovarianceAt(
110 ObjectState & state, const unsigned int i, const unsigned int j, const double val)
111{
112 sanityCheckContinuousStateCovariance(state);
113 const int n = getContinuousStateSize(state);
114 if (i >= static_cast<unsigned int>(n) || j >= static_cast<unsigned int>(n))
115 throw std::out_of_range("Covariance coordinates out of range");
116 if (state.continuous_state_covariance.empty()) {
117 setContinuousStateCovariance(state, std::vector<double>(n * n, CONTINUOUS_STATE_COVARIANCE_INIT));
118 for (int k = 0; k < n; ++k)
119 state.continuous_state_covariance[n * k + k] = CONTINUOUS_STATE_COVARIANCE_INVALID;
120 }
121 state.continuous_state_covariance.at(n * i + j) = val;
122}

346inline void setPoseCovariance(
347 ObjectState & state, const gm::PoseWithCovariance::_covariance_type & val)
348{
349 const int n = 6;
350 const int model_id = state.model_id;
351
352 int ix, jx;
353 for (int i = 0; i < n; i++) {
354 bool i_valid = true;
355 switch (i) {
356 case 0: i_valid = hasX(model_id); ix = i_valid ? indexX(model_id) : 0; break;
357 case 1: i_valid = hasY(model_id); ix = i_valid ? indexY(model_id) : 0; break;
358 case 2: i_valid = hasZ(model_id); ix = i_valid ? indexZ(model_id) : 0; break;
359 case 3: i_valid = hasRoll(model_id); ix = i_valid ? indexRoll(model_id) : 0; break;
360 case 4: i_valid = hasPitch(model_id); ix = i_valid ? indexPitch(model_id) : 0; break;
361 case 5: i_valid = hasYaw(model_id); ix = i_valid ? indexYaw(model_id) : 0; break;
362 }
363 if (!i_valid) continue;
364
365 for (int j = 0; j < n; j++) {
366 bool j_valid = true;
367 switch (j) {
368 case 0: j_valid = hasX(model_id); jx = j_valid ? indexX(model_id) : 0; break;
369 case 1: j_valid = hasY(model_id); jx = j_valid ? indexY(model_id) : 0; break;
370 case 2: j_valid = hasZ(model_id); jx = j_valid ? indexZ(model_id) : 0; break;
371 case 3: j_valid = hasRoll(model_id); jx = j_valid ? indexRoll(model_id) : 0; break;
372 case 4: j_valid = hasPitch(model_id); jx = j_valid ? indexPitch(model_id) : 0; break;
373 case 5: j_valid = hasYaw(model_id); jx = j_valid ? indexYaw(model_id) : 0; break;
374 }
375 if (!j_valid) continue;
376
377 setContinuousStateCovarianceAt(state, ix, jx, val[n * i + j]);
378 }
379 }

616inline void setVelocityXYZYawWithCovariance(
617 ObjectState & state, const gm::Vector3 & vel_xyz_in, const double yaw, const double var_vel_x,
618 const double var_vel_y, const double cov_vel_xy)
619{
620 gm::PoseWithCovariance vel_lon_lat, vel_xyz;
621 vel_xyz.pose.position.x = vel_xyz_in.x;
622 vel_xyz.pose.position.y = vel_xyz_in.y;
623 vel_xyz.pose.position.z = 0.0;
624
625 vel_xyz.covariance.at(0) = var_vel_x;
626 vel_xyz.covariance.at(1) = cov_vel_xy;
627 vel_xyz.covariance.at(6) = cov_vel_xy;
628 vel_xyz.covariance.at(7) = var_vel_y;
629
630 tf2::Quaternion q;
631 q.setRPY(0.0, 0.0, -yaw);
632 gm::TransformStamped tf;
633 tf.transform.rotation = tf2::toMsg(q);
634
635 gm::PoseWithCovarianceStamped vel_lon_lat_stamped, vel_xyz_stamped;
636 vel_xyz_stamped.pose = vel_xyz;
637 transformPoseWithCovariance(vel_xyz_stamped, vel_lon_lat_stamped, tf);
638 vel_lon_lat = vel_lon_lat_stamped.pose;
639
640 setVelocity(state, {vel_lon_lat.pose.position.x, vel_lon_lat.pose.position.y}, false);
641 setYaw(state, yaw, false);
642
643 int ix, jx, n = 2;
644 auto model_id = state.model_id;
645 for (int i = 0; i < n; i++) {
646 for (int j = 0; j < n; j++) {
647 if (i == 0 && hasVelLon(model_id))
648 ix = indexVelLon(model_id);
649 else if (i == 1 && hasVelLat(model_id))
650 ix = indexVelLat(model_id);
651 else
652 continue;
653
654 if (j == 0 && hasVelLon(model_id))
655 jx = indexVelLon(model_id);
656 else if (j == 1 && hasVelLat(model_id))
657 jx = indexVelLat(model_id);
658 else
659 continue;
660
661 setContinuousStateCovarianceAt(state, ix, jx, vel_lon_lat.covariance.at(6 * i + j));
662 }
663 }
664}
