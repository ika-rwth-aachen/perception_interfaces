78inline std::vector<double> getContinuousStateCovariance(const ObjectState& state) {
79 sanityCheckContinuousStateCovariance(state);
80 if (!state.continuous_state_covariance.empty()) return state.continuous_state_covariance;
81 const int n = getContinuousStateSize(state.model_id);
82 std::vector<double> covariance(n * n, CONTINUOUS_STATE_COVARIANCE_INIT);
83 for (int i = 0; i < n; ++i) covariance[n * i + i] = CONTINUOUS_STATE_COVARIANCE_INVALID;
84 return covariance;
85}

107inline double getContinuousStateCovarianceAt(const ObjectState& state, const unsigned int i, const unsigned int j) {
108 const int n = getContinuousStateSize(state);
109 if (i >= static_cast<unsigned int>(n) || j >= static_cast<unsigned int>(n))
110 throw std::out_of_range("Covariance coordinates out of range");
111 sanityCheckContinuousStateCovariance(state);
112 const auto& covariance = state.continuous_state_covariance;
113 if (covariance.empty()) return i == j ? CONTINUOUS_STATE_COVARIANCE_INVALID : CONTINUOUS_STATE_COVARIANCE_INIT;
114 return covariance.at(n * i + j);
115}

231inline gm::Point getCenterPosition(const ObjectState& state) {
232 gm::Point position = getPosition(state);
233 const auto orientation = getOrientation(state);
234 const gm::Vector3 offset_to_center = state.reference_point.translation_to_geometric_center;
235 tf2::Quaternion q;
236 tf2::fromMsg(orientation, q);
237 const tf2::Vector3 offset_to_center_tf2(offset_to_center.x, offset_to_center.y, offset_to_center.z);
238 const tf2::Vector3 rotated_offset_to_center = tf2::quatRotate(q, offset_to_center_tf2);
239 position.x += rotated_offset_to_center.x();
240 position.y += rotated_offset_to_center.y();
241 position.z += rotated_offset_to_center.z();
242 return position;
243}

275inline std::vector<double> getPoseCovariance(const ObjectState& state) {
276 const int n = 6;
277 const int model_id = state.model_id;
278 std::vector<double> pose_covariance(n * n, CONTINUOUS_STATE_COVARIANCE_INIT);
279 int ix, jx;
280 for (int i = 0; i < n; i++) {
281 for (int j = 0; j < n; j++) {
282 if (i == 0 && hasX(model_id))
283 ix = indexX(model_id);
284 else if (i == 1 && hasY(model_id))
285 ix = indexY(model_id);
286 else if (i == 2 && hasZ(model_id))
287 ix = indexZ(model_id);
288 else if (i == 3 && hasRoll(model_id))
289 ix = indexRoll(model_id);
290 else if (i == 4 && hasPitch(model_id))
291 ix = indexPitch(model_id);
292 else if (i == 5 && hasYaw(model_id))
293 ix = indexYaw(model_id);
294 else
295 continue;
296
297 if (j == 0 && hasX(model_id))
298 jx = indexX(model_id);
299 else if (j == 1 && hasY(model_id))
300 jx = indexY(model_id);
301 else if (j == 2 && hasZ(model_id))
302 jx = indexZ(model_id);
303 else if (j == 3 && hasRoll(model_id))
304 jx = indexRoll(model_id);
305 else if (j == 4 && hasPitch(model_id))
306 jx = indexPitch(model_id);
307 else if (j == 5 && hasYaw(model_id))
308 jx = indexYaw(model_id);
309 else
310 continue;
311
312 pose_covariance[n * i + j] = getContinuousStateCovarianceAt(state, ix, jx);
313 }
314 }
315 return pose_covariance;
316}

526inline gm::PoseWithCovariance getVelocityXYZWithCovariance(const ObjectState& state) {
527 gm::PoseWithCovariance vel_lon_lat, vel_xyz;
528 vel_lon_lat.pose.position.x = getVelLon(state);
529 vel_lon_lat.pose.position.y = getVelLat(state);
530 vel_lon_lat.pose.position.z = 0.0;
531
532 int ix, jx, n = 2;
533 auto model_id = state.model_id;
534 for (int i = 0; i < n; i++) {
535 for (int j = 0; j < n; j++) {
536 if (i == 0 && hasVelLon(model_id))
537 ix = indexVelLon(model_id);
538 else if (i == 1 && hasVelLat(model_id))
539 ix = indexVelLat(model_id);
540 else
541 continue;
542
543 if (j == 0 && hasVelLon(model_id))
544 jx = indexVelLon(model_id);
545 else if (j == 1 && hasVelLat(model_id))
546 jx = indexVelLat(model_id);
547 else
548 continue;
549
550 vel_lon_lat.covariance.at(6 * i + j) = getContinuousStateCovarianceAt(state, ix, jx);
551 }
552 }
553
554 tf2::Quaternion q;
555 q.setRPY(0.0, 0.0, getYaw(state));
556 gm::TransformStamped tf;
557 tf.transform.rotation = tf2::toMsg(q);
558
559 gm::PoseWithCovarianceStamped vel_lon_lat_stamped, vel_xyz_stamped;
560 vel_lon_lat_stamped.pose = vel_lon_lat;
561 transformPoseWithCovariance(vel_lon_lat_stamped, vel_xyz_stamped, tf);
562 vel_xyz = vel_xyz_stamped.pose;
563
564 return vel_xyz;
565}

782inline ObjectClassification getClassWithHighestProbability(const ObjectState& state) {
783 ObjectClassification highest_prob_class;
784 auto highest_prob_class_it = std::max_element(
785 state.classifications.begin(), state.classifications.end(),
786 [](const ObjectClassification& c1, const ObjectClassification& c2) { return c1.probability < c2.probability; });
787 if (highest_prob_class_it == state.classifications.end()) {
788 highest_prob_class.probability = -1.0;
789 highest_prob_class.type = ObjectClassification::UNKNOWN;
790 } else {
791 highest_prob_class = *highest_prob_class_it;
792 }
793 return highest_prob_class;
794}
